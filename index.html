-- ========================================== -- NIJI APORIA | DATABASE AUDIT -- READ ONLY - NO DATA WILL BE DELETED -- ========================================== -- 1. Existing public tables select table_name from information_schema.tables where table_schema = 'public' and table_type = 'BASE TABLE' order by table_name; -- 2. Existing users (counts only) select count(*) as total_auth_users from auth.users; select count(*) as total_profiles from public.profiles; -- 3. Existing user roles select role, count(*) as total from public.user_roles group by role order by role; -- 4. Check admin account without exposing email select exists ( select 1 from public.user_roles where role = 'admin' ) as admin_exists; -- 5. Row Level Security status select tablename, rowsecurity from pg_tables where schemaname = 'public' order by tablename; -- 6. Existing storage buckets select id, name, public from storage.buckets order by name; -- 7. Existing custom functions select routine_name from information_schema.routines where routine_schema = 'public' order by routine_name; -- 8. Existing triggers on user registration select trigger_name, event_object_table from information_schema.triggers where event_object_schema = 'auth' and event_object_table = 'users';
